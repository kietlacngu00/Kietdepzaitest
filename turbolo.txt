local function SaveSettings()

local function LoadSettings()
    if not (isfile and isfile(FullPath)) then return false end
    local success, result = pcall(function()
        return HttpService:JSONDecode(readfile(FullPath))
    end)
    if success and result then _G.SaveData = result return true end
    return false
end

local function HasKen()
    local char = player.Character
    return char and CollectionService:HasTag(char, "Ken")
end

local function EquipWeapon(weaponName)
    if not weaponName then return end
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local tool = player.Backpack:FindFirstChild(weaponName) or char:FindFirstChild(weaponName)
    if tool and tool.Parent ~= char then
        hum:EquipTool(tool)
    end
end

local function UseSkill(category, key)
    EquipWeapon(_G.SelectWeapon)
    local keys = {Z = "Z", X = "X", C = "C", V = "V", F = "F"}
    if keys[key] then
        VirtualInputManager:SendKeyEvent(true, keys[key], false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, keys[key], false, game)
    end
end

local function IsAlive(entity)
    local hum = entity and entity:FindFirstChild("Humanoid")
    return hum and hum.Health > 0
end

local function GetClosestEnemy(enemyList, maxDistance)
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    
    local closest, shortest = nil, math.huge
    for _, enemy in pairs(enemyList) do
        if IsAlive(enemy) then
            local root = enemy:FindFirstChild("HumanoidRootPart")
            if root then
                local dist = (root.Position - hrp.Position).Magnitude
                if dist < shortest and dist <= maxDistance then
                    shortest = dist
                    closest = enemy
                end
            end
        end
    end
    return closest
end

local function TeleportTo(targetCFrame)
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp and targetCFrame then
        hrp.CFrame = targetCFrame
    end
end

local function GetCurrentQuestData()
    local level = player.Data.Level.Value
    local bestLevel = 0
    local bestData = nil
    
    for reqLevel, data in pairs(QuestData) do
        if level >= reqLevel and reqLevel > bestLevel then
            bestLevel = reqLevel
            bestData = data
        end
    end
    
    return bestData
end

local currentTarget = nil
local lastAttackTime = 0
local ATTACK_COOLDOWN = 0.15

task.spawn(function()
    while _G.StartFarm and _G.Level do
        pcall(function()
            local char = player.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if not hrp then task.wait(1) return end
            
            local questData = GetCurrentQuestData()
            if not questData then task.wait(1) return end
            
            local questUI = player.PlayerGui:FindFirstChild("Main") and player.PlayerGui.Main:FindFirstChild("Quest")
            local hasQuest = questUI and questUI.Visible
            
            -- Accept quest
            if _G.AcceptQuest and not hasQuest then
                if (hrp.Position - questData.PosQ.Position).Magnitude > 10 then
                    TeleportTo(questData.PosQ)
                else
                    task.wait(0.5)
                    commF:InvokeServer("StartQuest", questData.Qname, questData.Qdata)
                end
                return
            end
            
            -- Find and attack enemy
            if not currentTarget or not IsAlive(currentTarget) then
                currentTarget = GetClosestEnemy(Workspace.Enemies:GetChildren(), 250)
            end
            
            if currentTarget and IsAlive(currentTarget) then
                local root = currentTarget:FindFirstChild("HumanoidRootPart")
                if root then
                    if (hrp.Position - root.Position).Magnitude > 15 then
                        TeleportTo(root.CFrame * CFrame.new(0, _G.MobHeight, 0))
                    end
                    
                    if tick() - lastAttackTime >= ATTACK_COOLDOWN then
                        UseSkill(_G.SelectWeapon, "Z")
                        UseSkill(_G.SelectWeapon, "X")
                        UseSkill(_G.SelectWeapon, "C")
                        lastAttackTime = tick()
                    end
                end
            else
                TeleportTo(questData.PosM)
            end
        end)
        task.wait(0.1)
    end
end)

-- BRING MOBS SYSTEM (Optimized)
-- ========================================
local bringEnabled = GetSetting("BringMobs", true)
local tweenInfo = TweenInfo.new(0.5, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)

local function BringEnemies(targetPos)
    if not bringEnabled or not _G.StartFarm then return end
    
    local count = 0
    for _, mob in pairs(Workspace.Enemies:GetChildren()) do
        if count >= _G.MaxBringMobs then break end
        
        local hum = mob:FindFirstChild("Humanoid")
        local root = mob:FindFirstChild("HumanoidRootPart")
        
        if hum and root and hum.Health > 0 and not root:GetAttribute("Tweening") then
            local dist = (root.Position - targetPos).Magnitude
            if dist <= _G.BringRange then
                count = count + 1
                root:SetAttribute("Tweening", true)
                
                local tween = TweenService:Create(root, tweenInfo, {CFrame = CFrame.new(targetPos)})
                tween:Play()
                tween.Completed:Once(function()
                    if root then root:SetAttribute("Tweening", false) end
                end)
            end
        end
    end
end

task.spawn(function()
    while true do
        if bringEnabled and _G.StartFarm then
            local char = player.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if hrp then
                BringEnemies(hrp.Position)
            end
        end
        task.wait(1)
    end
end)

local autoAttackEnabled = GetSetting("AutoAttack", true)

local function GetNearbyEnemies()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return {} end
    
    local enemies = {}
    for _, mob in pairs(Workspace.Enemies:GetChildren()) do
        if IsAlive(mob) then
            local root = mob:FindFirstChild("HumanoidRootPart")
            if root and (root.Position - hrp.Position).Magnitude <= 60 then
                table.insert(enemies, {mob, root})
            end
        end
    end
    return enemies
end

local registerAttack = net:WaitForChild("RE/RegisterAttack")
local registerHit = net:WaitForChild("RE/RegisterHit")

task.spawn(function()
    while autoAttackEnabled and _G.StartFarm do
        local enemies = GetNearbyEnemies()
        if #enemies > 0 then
            pcall(function()
                registerAttack:FireServer(-math.huge)
                local hitData = {nil, {}}
                for i, data in ipairs(enemies) do
                    if not hitData[1] then hitData[1] = data[1].Head end
                    hitData[2][i] = data
                end
                registerHit:FireServer(unpack(hitData))
            end)
        end
        task.wait(0.05)
    end
end)

local function LowCPU()
    Workspace.Terrain.WaterWaveSize = 0
    Workspace.Terrain.WaterWaveSpeed = 0
    Workspace.Terrain.WaterReflectance = 0
    Workspace.Terrain.WaterTransparency = 0
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e9
    settings().Rendering.QualityLevel = "Level01"
    
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("Part") or v:IsA("Union") then
            v.Material = "Plastic"
            v.Reflectance = 0
        elseif v:IsA("ParticleEmitter") or v:IsA("Trail") then
            v.Lifetime = NumberRange.new(0)
        elseif v:IsA("Fire") or v:IsA("SpotLight") or v:IsA("Smoke") then
            v.Enabled = false
        end
    end
    
    for _, v in pairs(Lighting:GetChildren()) do
        if v:IsA("BlurEffect") or v:IsA("SunRaysEffect") or v:IsA("BloomEffect") then
            v.Enabled = false
        end
    end
end

LowCPU()
