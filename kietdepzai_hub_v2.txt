-- HÀM KÉO THẢ (DRAGGABLE) MƯỢT MÀ CHO MỌI THIẾT BỊ
local function MakeDraggable(dragPart, mainGfx)
    local dragging, dragInput, mousePos, framePos
    dragPart.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            mousePos = input.Position
            framePos = mainGfx.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    dragPart.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - mousePos
            mainGfx.Position = UDim2.new(framePos.X.Scale, framePos.X.Offset + delta.X, framePos.Y.Scale, framePos.Y.Offset + delta.Y)
        end
    end)
end

local function UpdateTimeAndGreeting()
    local timeTbl = os.date("*t")
    local h = timeTbl.hour
    local m = timeTbl.min
    local s = timeTbl.sec
    local day = timeTbl.day
    local month = timeTbl.month
    local year = timeTbl.year
    
    local greeting = ""
    if h >= 5 and h <= 10 then greeting = "CHÀO BUỔI SÁNG"
    elseif h == 11 or h == 12 then greeting = "CHÀO BUỔI TRƯA"
    elseif h >= 13 and h <= 17 then greeting = "CHÀO BUỔI CHIỀU"
    else greeting = "CHÀO BUỔI TỐI" end
    
    TitleLabel.Text = "🍊 KIETDEPZAI HUB - " .. greeting .. "!"
    return string.format("%02d:%02d:%02d | %02d/%02d/%04d", h, m, s, day, month, year)
end

-- ==================== CÁC HIỆU ỨNG RANDOM KHI CLICK (DÀNH CHO NÚT CHỌN) ====================
local RandomEffects = {
    function(btn) -- Lắc nhẹ nút (Shake)
        local origPos = btn.Position
        local tw1 = TweenService:Create(btn, TweenInfo.new(0.05), {Position = origPos + UDim2.new(0,5,0,0)})
        local tw2 = TweenService:Create(btn, TweenInfo.new(0.05), {Position = origPos - UDim2.new(0,5,0,0)})
        local tw3 = TweenService:Create(btn, TweenInfo.new(0.05), {Position = origPos})
        tw1:Play(); tw1.Completed:Wait(); tw2:Play(); tw2.Completed:Wait(); tw3:Play()
    end,
    function(btn) -- Nháy sáng đổi màu (Flash Color)
        local origColor = btn.BackgroundColor3
        btn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        TweenService:Create(btn, TweenInfo.new(0.3), {BackgroundColor3 = Color3.fromRGB(255, 102, 0)}):Play()
    end,
    function(btn) -- Xoay một vòng (Spin)
        local tw = TweenService:Create(btn, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Rotation = 360})
        tw:Play()
        tw.Completed:Connect(function() btn.Rotation = 0 end)
    end,
    function(btn) -- Bật lớn rồi thu nhỏ (Pop/Scale)
        local origSize = btn.Size
        local tw1 = TweenService:Create(btn, TweenInfo.new(0.1), {Size = origSize + UDim2.new(0,10,0,6)})
        tw1:Play(); tw1.Completed:Wait()
        TweenService:Create(btn, TweenInfo.new(0.15), {Size = origSize}):Play()
    end
}
