-- THÔNG BÁO NHẢY LÊN (Custom Toast Notification System)
local function CreateNotification(text)
    local NotifFrame = Instance.new("Frame")
    local NotifCorner = Instance.new("UICorner")
    local NotifText = Instance.new("TextLabel")
    
    NotifFrame.Size = UDim2.new(0, 240, 0, 40)
    NotifFrame.Position = UDim2.new(1, 20, 0.85, 0) -- Xuất phát ngoài màn hình bên phải
    NotifFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    NotifFrame.BorderColor3 = Color3.fromRGB(255, 102, 0)
    NotifFrame.BorderSizePixel = 1
    NotifFrame.Parent = ScreenGui
    
    NotifCorner.CornerRadius = UDim.new(0, 6)
    NotifCorner.Parent = NotifFrame
    
    NotifText.Size = UDim2.new(1, 0, 1, 0)
    NotifText.BackgroundTransparency = 1
    NotifText.Font = Enum.Font.GothamBold
    NotifText.TextColor3 = Color3.fromRGB(255, 102, 0) -- Màu cam nổi bật
    NotifText.TextSize = 13
    NotifText.Text = text
    NotifText.Parent = NotifFrame
    
    -- Animation bay vào từ bên phải
    NotifFrame:TweenPosition(UDim2.new(1, -260, 0.85, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quart, 0.4, true)
    
    -- Tự động bay đi mất sau 2.5 giây
    task.delay(2.5, function()
        NotifFrame:TweenPosition(UDim2.new(1, 20, 0.85, 0), Enum.EasingDirection.In, Enum.EasingStyle.Quart, 0.4, true, function()
            NotifFrame:Destroy()
        end)
    end)
end
